cmake_minimum_required(VERSION 3.16)
project(DA030 LANGUAGES CXX)
if(NOT WIN32)
    message(FATAL_ERROR "DA030 uses the Windows API and requires Windows.")
endif()
add_executable(DA030 WIN32 Main.cpp)
target_compile_features(DA030 PRIVATE cxx_std_17)
target_link_libraries(DA030 PRIVATE comctl32 gdi32 user32)
if(MSVC)
    target_compile_options(DA030 PRIVATE /W4 /permissive-)
else()
    target_compile_options(DA030 PRIVATE -Wall -Wextra)
endif()
